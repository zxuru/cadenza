# Flet 1.0 UI API (ft.run, ft.Colors, ft.Icons, page.window, ...), plus the
# desktop client `flet_desktop`: the frozen app launches it as a separate
# process, and build.py stages its archive into the executable. `flet_web` is
# deliberately not asked for - it is only used by the browser view.
flet[desktop]>=1.0,<2

# Engine: extractor fixes land constantly, so leave this unpinned and update often.
yt-dlp

# Tags and cover art: yt-dlp writes the picture into flac files with this (mp3
# and m4a go through ffmpeg), and `metadata.py` imports it to write the tags a
# music database gave us. Without it a flac download fails outright.
mutagen

# Optional fallback: ships a static ffmpeg binary, used only when `ffmpeg`
# is not on PATH. Drop this line if your system already has ffmpeg.
imageio-ffmpeg>=0.6

# Not installed here: `av` and `pillow`, which convert audio in-process when
# there is no ffmpeg binary at all - the Android build's own path (see
# transcode.py and pyproject.toml). On a desktop they are only needed if
# neither the system ffmpeg nor imageio-ffmpeg is available.
